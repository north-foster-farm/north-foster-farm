{{/* A captioned photo from assets/images, in the home page's frame. Hugo's built-in figure prints `src` as written, which for a file under assets/ is a path the browser cannot fetch. This one resolves the file as a resource, resizes it to the width the home page uses, and hands it to home/figure.html. {{< figure src="images/lily.jpg" alt="…" caption="…" />}} A caption that runs long goes inside the pair instead, where it can wrap like prose: {{< figure src="images/lily.jpg" alt="…" >}} Caption text. {{< /figure >}} Optional: shape="egg" clips a portrait photo to the egg and sets it beside the text; class adds classes to the figure; frame="false" drops the rounding and shadow, for a cut-out such as a PNG on transparency. */}} {{ $src := .Get "src" }} {{ $image := resources.Get $src }} {{ if not $image }} {{ errorf "figure: no image at assets/%s (%s)" $src .Position }} {{ end }} {{ $caption := .Get "caption" }} {{ with .Inner }}{{ $caption = . | markdownify }}{{ end }} {{ $shape := .Get "shape" | default "" }} {{ $width := cond (eq $shape "egg") "600x" "1150x" }} {{ partial "home/figure.html" (dict "image" ($image.Resize (printf "%s webp q82" $width)) "alt" (.Get "alt") "caption" $caption "shape" $shape "frame" (ne (.Get "frame") "false") "classes" (printf "my-4 %s" (.Get "class" | default "") | strings.TrimSpace)) }}
