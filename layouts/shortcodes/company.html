{{- /* The farm's email or phone from data/company.json as a Markdown link, so a page never restates them (Q21b). Call it with % so the link renders in the page and in its Markdown output: {{% company "email" %}} [sales@…](mailto:sales@…) {{% company "email-info" %}} [info@…](mailto:info@…) {{% company "phone" %}} [(401) …](tel:+1401…) A second argument is the link's text instead: {{% company "email" "email us" %}} "email" is sales@, for anything commercial (ordering, orders, pickup, delivery, order problems). "email-info" is info@, for anything non-commercial (privacy, accessibility, data requests, press, legal). */ -}} {{- $company := hugo.Data.company -}} {{- $fact := .Get 0 -}} {{- $link := "" -}} {{- if eq $fact "email" -}} {{- $link = printf "[%s](mailto:%s)" (.Get 1 | default $company.email) $company.email -}} {{- else if eq $fact "email-info" -}} {{- $link = printf "[%s](mailto:%s)" (.Get 1 | default $company.emailInfo) $company.emailInfo -}} {{- else if eq $fact "phone" -}} {{- $link = printf "[%s](tel:%s)" (.Get 1 | default $company.phone.display) $company.phone.plain -}} {{- else -}} {{- errorf "company: no fact %q in %s" $fact .Position -}} {{- end -}} {{- $link | safeHTML -}}
