{{/* The in-stock catalog groups, for the sidebar and the product list. A group with nothing to sell is left out, and so is its link. A group's `unit` ("per pack") sits by its heading; an item's own `unit` sits in its row, for the groups that mix them. */}} {{ $groups := slice }} {{ range hugo.Data.catalog.groups }} {{ $items := where .items "inStock" true }} {{ if $items }} {{ $short := replaceRE ` \(.*\)$` "" .label }} {{ $groups = $groups | append (dict "key" .key "label" .label "short" $short "unit" (.unit | default "") "items" $items) }} {{ end }} {{ end }} {{ return $groups }}
