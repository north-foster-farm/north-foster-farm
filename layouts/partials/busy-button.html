{{/* A button that fades to an egg spinner and a busy word while its request runs (#299: every button that sends one). Both faces share one grid cell, so the button keeps the width of the wider one and only fades between them. scripts/utils/busy-button.js turns it on and off. Params: label The idle text busy The text beside the spinner class Extra classes, such as btn-primary (optional) id The button's id (optional) type The button's type (optional; submit if not given) disabled true to start disabled (optional) hidden true to start hidden (optional) fit true to fit the face that's showing, easing between the two widths, rather than keep the wider (optional) attrs More attributes, as a dict; an empty value writes the name alone, such as data-close (optional) plain true to leave out Bootstrap's btn, for a button its own class styles, such as a menu item (optional) start true to set both faces at the start, not centered, as in a menu (optional) icon An SVG under assets/images to show before the label, the spinner taking its place while busy (optional) iconClass The class of the span that holds the icon (optional) */}} {{- $class := slice -}} {{- if not .plain }}{{ $class = $class | append "btn" }}{{ end -}} {{- $class = $class | append "busy-button" -}} {{- if .fit }}{{ $class = $class | append "busy-button-fit" }}{{ end -}} {{- if .start }}{{ $class = $class | append "busy-button-start" }}{{ end -}} {{- with .class }}{{ $class = $class | append . }}{{ end -}} {{- $attrs := "" -}} {{- range $name, $value := .attrs -}} {{- if eq (string $value) "" -}} {{- $attrs = printf "%s %s" $attrs $name -}} {{- else -}} {{- $attrs = printf "%s %s=%q" $attrs $name (string $value | htmlEscape) -}} {{- end -}} {{- end -}} {{- $spinner := partial "svg.html" (dict "src" "icons/egg-spinner" "class" "egg-spinner" "aria-hidden" "true" "focusable" "false") -}}
