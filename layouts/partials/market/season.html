{{/* A market's season in words, from its "opens" and "closes" dates in data/markets.json, so the words can't disagree with the dates: "June through September". */}} {{ $zone := "America/New_York" }} {{ $opens := (time.AsTime .opens $zone).Format "January" }} {{ $closes := (time.AsTime .closes $zone).Format "January" }} {{ return cond (eq $opens $closes) $opens (printf "%s through %s" $opens $closes) }}
